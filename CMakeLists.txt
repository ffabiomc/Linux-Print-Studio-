cmake_minimum_required(VERSION 3.20)
project(LinuxPrintStudio VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4)
pkg_check_modules(ADWAITA REQUIRED IMPORTED_TARGET libadwaita-1)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-glib)
pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
pkg_check_modules(CUPS REQUIRED IMPORTED_TARGET cups)
add_library(print_core src/core.cpp)
target_include_directories(print_core PUBLIC include)
target_link_libraries(print_core PUBLIC PkgConfig::POPPLER PkgConfig::CAIRO PkgConfig::CUPS)
target_compile_options(print_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(linux-print-studio src/main.cpp)
target_link_libraries(linux-print-studio PRIVATE print_core PkgConfig::GTK PkgConfig::ADWAITA)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE print_core)
add_test(NAME core_tests COMMAND core_tests)
install(TARGETS linux-print-studio RUNTIME DESTINATION bin)
